::consonant:: = b|d|ch|f|g|h|j|k|l|m|n|p|r|s|t|th|v|w|z
::vowel:: = a|e|i|o|u

% Realization of coda <g>/<gh>
g -> ch / (::vowel::) _ #|(::consonant::)
gh -> ch / (::vowel::) _ #|(::consonant::)

% Final stop devoicing
b -> p / (::vowel::)(?:(::consonant::))? _ #|(::consonant::)
d -> t / (::vowel::)(?:(::consonant::))? _ #|(::consonant::)

% Contextual pronunciation of <sch>
sch -> s / _ #|((?!r)(::consonant::))

% Realization of <c>
c -> k / _ #
c -> s / _ [eiy]

% Open syllable lengthening
a -> aa / (?<![aeiou]) _ #|(::consonant::)(::vowel::)
e -> ee / (?<![aeiou]) _ #|(::consonant::)(::vowel::)
i -> ii / (?<![aeiou]) _ (?!j)(#|(::consonant::)(::vowel::))
o -> oo / (?<![aeiou]) _ #|(::consonant::)(::vowel::)
u -> uu / (?<![aeiou]) _ #|(::consonant::)(::vowel::)
